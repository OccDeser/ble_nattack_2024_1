    \node[ORACLE] (n_oracle) {
        \hspace{0em}\textbf{if} $x_i \in X$ \& $x_i = a_i$: \\
        \hspace{1.5em}\textbf{if} $\forall k, f'(X \setminus \{x_i\}, k) = f(X, k)$:\\
        \hspace{3em}\textbf{return} $f'(X \setminus \{x_i\}, k)$
    };
    \node[anchor=north east] at (n_oracle.north east) {RO};

    \coordinate (input1) at ($(n_oracle.west)+(-13.5em, 4.5ex)$);
    \coordinate (input1_end) at ($(input1-|n_oracle.west)$);
    \draw[ARROW] (input1) -- (input1_end) node[midway,above=-0.3ex]{$f(X, k)$, $X = \{x_1, x_2, \ldots, x_n\}$};

    \coordinate (input2) at ($(n_oracle.west)+(-13.5em, 0.5ex)$);
    \coordinate (input2_end) at ($(input2-|n_oracle.west)$);
    \draw[ARROW] (input2) -- (input2_end) node[midway,above=-0.3ex]{$a_i$};

    \coordinate (output) at ($(n_oracle.west)+(-13.5em, -5ex)$);
    \coordinate (output_end) at ($(output-|n_oracle.west)$);
    \draw[ARROW] (output_end) -- (output) node[midway,above=-0.3ex]{$f'(X \setminus \{x_i\}, k)$};
